    private static function echoChanges(string $markdown, string $label): void
    {
        $section = Desk::latestSection($markdown);
        if ($section === '') {
            return;
        }
        echo '<h4>'.esc_html($label).'</h4>';
        self::echoMarkdown($section);
    }

    private static function echoNotice(string $notice, string $reference, string $supportEmail): void
    {
        $shown = $reference !== '' ? $reference : __('your ticket', 'theme-desk');
        if ($notice === 'ticket-sent') {
            echo '<div class="notice notice-success inline" role="status"><p><strong>'.esc_html(sprintf(
                /* translators: %s: ticket reference. */
                __('Ticket %s is on its way.', 'theme-desk'),
                $shown
            )).'</strong> '.esc_html(sprintf(
                /* translators: %s: support email address. */
                __('It was emailed to %s. Replies come to the address you gave us. Quote the ticket number if you write again.', 'theme-desk'),
                $supportEmail
            )).'</p></div>';

            return;
        }
        if ($notice === 'ticket-failed') {
            echo '<div class="notice notice-warning inline" role="alert"><p><strong>'.esc_html(sprintf(
                /* translators: %s: ticket reference. */
                __('Ticket %s is saved, but the email could not be sent from this site.', 'theme-desk'),
                $shown
            )).'</strong> '.esc_html__('This usually means the site has no working email setup. Use Send again in the list below once it is fixed, or email us yourself with the link there.', 'theme-desk').'</p></div>';

            return;
        }
        if ($notice === 'ticket-wait') {
            echo '<div class="notice notice-warning inline" role="alert"><p>'.esc_html__('Please wait a few seconds before sending another ticket.', 'theme-desk').'</p></div>';

            return;
        }
        if ($notice === 'ticket-hour') {
            echo '<div class="notice notice-warning inline" role="alert"><p>'.esc_html__('You have reached the hourly limit for tickets from this site. Please wait a while, or reply to the email from your last ticket to add details.', 'theme-desk').'</p></div>';

            return;
        }
        if ($notice === 'ticket-error') {
            echo '<div class="notice notice-error inline" role="alert"><p>'.esc_html__('The ticket could not be saved. Please try again.', 'theme-desk').'</p></div>';

            return;
        }
        if ($notice === 'ticket-removed') {
            echo '<div class="notice notice-success inline" role="status"><p>'.esc_html__('The ticket was removed from this list.', 'theme-desk').'</p></div>';
        }
    }

    /**
     * @param  array<string, string>  $problems
     */
    private static function fieldAttrs(string $key, array $problems, bool $hasHelp = false): void
    {
        $ids = [];
        if ($hasHelp) {
            $ids[] = 'theme-desk-'.$key.'-help';
        }
        if (isset($problems[$key])) {
            $ids[] = 'theme-desk-'.$key.'-error';
            echo ' aria-invalid="true"';
        }
        if ($ids !== []) {
            echo ' aria-describedby="'.esc_attr(implode(' ', $ids)).'"';
        }
    }
