$problems[$key])
        );
    }

    private static function echoMarkdown(string $markdown): void
    {
        $inList = false;
        $paragraph = [];
        $lines = preg_split("/\\n/", str_replace(["\\r\\n", "\\r"], "\\n", $markdown));
        if (! is_array($lines)) {
            return;
        }
        foreach ($lines as $line) {
            $trim = trim($line);
            if ($trim === '') {
                self::echoParagraph($paragraph);
                $paragraph = [];
                $inList = self::closeList($inList);

                continue;
            }
            if (str_starts_with($trim, '# ')) {
                self::echoParagraph($paragraph);
                $paragraph = [];
                $inList = self::closeList($inList);

                continue;
            }
            if (str_starts_with($trim, '### ') || str_starts_with($trim, '## ')) {
                self::echoParagraph($paragraph);
                $paragraph = [];
                $inList = self::closeList($inList);
                $heading = str_starts_with($trim, '### ') ? substr($trim, 4) : substr($trim, 3);
                echo '<h4>';
                self::echoInline($heading);
                echo '</h4>';

                continue;
            }
            if (str_starts_with($trim, '- ')) {
                self::echoParagraph($paragraph);
                $paragraph = [];
                if (! $inList) {
                    echo '<ul>';
                    $inList = true;
                }
                echo '<li>';
                self::echoInline(substr($trim, 2));
                echo '</li>';

                continue;
            }
            $inList = self::closeList($inList);
            $paragraph[] = $trim;
        }
        self::echoParagraph($paragraph);
        self::closeList($inList);
    }

    /**
     * @param  list<string>  $paragraph
     */
    private static function echoParagraph(array $paragraph): void
    {
        if ($paragraph === []) {
            return;
        }
        echo '<p>';
        self::echoInline(implode(' ', $paragraph));
        echo '</p>';
    }

    private static function closeList(bool $open): bool
    {
        if ($open) {
            echo '</ul>';
        }

        return false;
    }

    private static function echoInline(string $text): void
    {
        $parts = preg_split('/\\*\\*(.+?)\\*\\*/', $text, -1, PREG_SPLIT_DELIM_CAPTURE);
        if (! is_array($parts)) {
            echo esc_html($text);

            return;
        }
        foreach ($parts as $index => $part) {
            if ($index % 2 === 1) {
                echo '<strong>'.esc_html($part).'</strong>';

                continue;
            }
            echo esc_html($part);
        }
    }
